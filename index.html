# -*- coding: utf-8 -*- """ Extrae SOLO el cráneo y las vértebras C1 a C7 de Z-Anatomy y las exporta como un único archivo GLB: cuello_cabeza.glb Piezas que quedan en el GLB (nombres que lee el visor): SKULL, C1, C2, C3, C4, C5, C6, C7 USO (Blender 3.6 o superior; recomendado 4.x): 1. Abre el archivo .blend de Z-Anatomy. 2. Pestaña "Scripting" > Text > Open > elige este archivo. 3. Presiona "Run Script" (triángulo). 4. Revisa el texto "LOG_EXPORTACION" (menú desplegable del editor de texto). 5. Al terminar, CIERRA Blender SIN GUARDAR (el script modifica la escena en memoria). """ import bpy import os import re # ---------------------------------------------------------------------------- # CONFIGURACIÓN # ---------------------------------------------------------------------------- OUT = os.path.join(os.path.expanduser("~"), "cuello_cabeza.glb") # carpeta de usuario RATIO_DECIMAR = 0.5 # 1.0 = sin reducir. 0.5 = deja la mitad de polígonos (GLB más liviano) # Objetos cuyo nombre contenga alguno de estos términos NO se usan EXCLUIR = re.compile( r"nerve|ganglion|muscle|ligament|disc|disk|artery|vein|vessel|root|membrane|fascia|" r"plexus|branch|joint|capsule|sheath|cord|dura|menin|segment|sinus|gland|label|brain|" r"lobe|tendon|cartilage|fat|skin" ) ROM = {3: "iii", 4: "iv", 5: "v", 6: "vi", 7: "vii"} ORD = {3: "third|3rd", 4: "fourth|4th", 5: "fifth|5th", 6: "sixth|6th", 7: "seventh|7th"} def patron_cervical(n): r = ROM[n] extra = r"|vertebra prominens" if n == 7 else "" return re.compile( rf"\bc ?({n}|{r})\b|cervical\w* (vertebra\w* )?\(?({n}|{r})\b|" rf"vertebra\w* cervical\w* ({n}|{r})\b|({ORD[n]}) cervical{extra}" ) # Si algún patrón no encuentra su pieza, el script lo informa y se ajusta aquí. PATRONES = { "C1": re.compile(r"\batlas\b|\bc ?(1|i)\b"), "C2": re.compile(r"\baxis\b|\bc ?(2|ii)\b|\bdens\b|odontoid"), "C3": patron_cervical(3), "C4": patron_cervical(4), "C5": patron_cervical(5), "C6": patron_cervical(6), "C7": patron_cervical(7), "SKULL": re.compile( r"\bskull\b|cranium|calvaria|frontal bone|parietal bone|occipital bone|temporal bone|" r"sphenoid bone|ethmoid bone|mandible|maxilla|zygomatic bone|nasal bone|lacrimal bone|" r"\bvomer\b|palatine bone|nasal concha|\btooth\b|\bteeth\b" ), } LOG = [] def log(*a): s = " ".join(str(x) for x in a) print(s) LOG.append(s) def normalizar(nombre): n = nombre.lower() n = re.sub(r"[\[\]\(\)_\-\.,/]", " ", n) return re.sub(r"\s+", " ", n).strip() # ---------------------------------------------------------------------------- # 1. BUSCAR PIEZAS # ---------------------------------------------------------------------------- encontradas = {k: [] for k in PATRONES} for o in bpy.data.objects: if o.type != "MESH": continue if o.name.endswith((".t", ".j")): continue n = normalizar(o.name) if EXCLUIR.search(n): continue for clave, pat in PATRONES.items(): if pat.search(n): encontradas[clave].append(o) break log("=== PIEZAS ENCONTRADAS ===") faltan = [] for clave, lista in encontradas.items(): log(f"{clave}: {len(lista)} objeto(s)") for o in lista[:10]: log(" -", o.name) if len(lista) > 10: log(f" ... y {len(lista) - 10} más") if not lista: faltan.append(clave) if faltan: log("") log("NO SE EXPORTÓ: faltan piezas ->", ", ".join(faltan)) log("Pistas: nombres de objetos que contienen 'vert', 'atlas', 'axis', 'cervic' o 'skull':") pistas = [o.name for o in bpy.data.objects if o.type == "MESH" and re.search(r"vert|atlas|axis|cervic|skull|crani", o.name.lower())] for nombre in sorted(pistas)[:80]: log(" ?", nombre) log("Ajusta PATRONES en este script según esos nombres y vuelve a ejecutarlo.") t = bpy.data.texts.new("LOG_EXPORTACION") t.write("\n".join(LOG)) raise SystemExit("Faltan piezas: ver texto LOG_EXPORTACION") # ---------------------------------------------------------------------------- # 2. ESCENA TEMPORAL CON COPIAS (no toca la escena original) # ---------------------------------------------------------------------------- tmp = bpy.data.scenes.new("EXPORT_CUELLO") bpy.context.window.scene = tmp vl = tmp.view_layers[0] col = tmp.collection resultado = {} for clave, lista in encontradas.items(): copias = [] for k, o in enumerate(lista): me = o.data.copy() c = bpy.data.objects.new(f"{clave}_{k}", me) c.matrix_world = o.matrix_world.copy() # conserva la posición en el mundo col.objects.link(c) copias.append(c) if len(copias) > 1: with bpy.context.temp_override(scene=tmp, view_layer=vl, active_object=copias[0], object=copias[0], selected_objects=copias, selected_editable_objects=copias): bpy.ops.object.join() obj = copias[0] obj.name = clave obj.data.name = clave obj.data.materials.clear() for p in obj.data.polygons: p.use_smooth = True if RATIO_DECIMAR < 1.0: try: m = obj.modifiers.new("dec", "DECIMATE") m.ratio = RATIO_DECIMAR with bpy.context.temp_override(scene=tmp, view_layer=vl, active_object=obj, object=obj): bpy.ops.object.modifier_apply(modifier=m.name) except Exception as e: log("Aviso: no se pudo reducir", clave, "->", e) resultado[clave] = obj # Quitar las demás escenas para que el GLB contenga solo estas piezas for sc in list(bpy.data.scenes): if sc is not tmp: try: bpy.data.scenes.remove(sc) except Exception as e: log("Aviso: no se pudo quitar la escena", sc.name, "->", e) # ---------------------------------------------------------------------------- # 3. EXPORTAR # ---------------------------------------------------------------------------- kwargs = dict(filepath=OUT, export_format="GLB", export_yup=True, export_apply=True, use_active_scene=True) try: bpy.ops.export_scene.gltf(**kwargs) except TypeError: kwargs.pop("use_active_scene") bpy.ops.export_scene.gltf(**kwargs) tam = os.path.getsize(OUT) / (1024 * 1024) log("") log(f"LISTO: {OUT} ({tam:.1f} MB)") log("Piezas exportadas:", ", ".join(resultado.keys())) log("Cierra Blender SIN guardar.") t = bpy.data.texts.new("LOG_EXPORTACION") t.write("\n".join(LOG))
